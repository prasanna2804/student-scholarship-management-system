// File upload handling function
function uploadFile($input_name, $target_dir) {
    $file_name = basename($_FILES[$input_name]["name"]);
    $file_type = strtolower(pathinfo($file_name, PATHINFO_EXTENSION));
    
    if (!in_array($file_type, ["jpg", "jpeg", "png", "pdf"])) {
        return null; // Invalid file type
    }

    $new_name = uniqid() . "_" . $file_name; // Rename to prevent overwriting
    $target_file = $target_dir . $new_name;

    if (move_uploaded_file($_FILES[$input_name]["tmp_name"], $target_file)) {
        return $target_file; // Return file path
    }

    return null;
}
